void remove_alpha(char a[],char s[])
{
	for(int i=0;a[i]!=0;i++)
	{
		for(int j=0;s[j]!=0;j++)
		{
			if(a[i]==s[j] && a[i]!=' ')
			{
				for(int k=i;a[k]!=0;k++)
				{
					a[k]=a[k+1];
				}
				i--;
				break;
			}
		}
	}
}
